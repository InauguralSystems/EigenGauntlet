   383	/* Shannon entropy over a NUL-terminated byte string (256-bin frequency count).
   384	 * Shared by VAL_STR and VAL_JSON_RAW: both hold `data.str`, so the same bytes
   385	 * must measure the same either way. VAL_JSON_RAW used to return a flat 0.0 —
   386	 * a document could grow without bound and the observer still reported it as
   387	 * `equilibrium` with dH exactly 0 forever. */
   388	static double entropy_of_cstr(const char *sp) {
   389	    if (!sp || !sp[0]) return 0.0;
   390	    int freq[256] = {0};
   391	    int len = 0;
   392	    for (const char *c = sp; *c; c++) { freq[(unsigned char)*c]++; len++; }
   393	    if (len == 0) return 0.0;
   394	    double h = 0.0;
   395	    for (int i = 0; i < 256; i++) {
   396	        if (freq[i] > 0) { double p = (double)freq[i] / len; h -= p * log2(p); }
   397	    }
   398	    return h;
   399	}

   403	/* A container child contributes only its own size term: the walk STOPS at a
   404	 * reference. This is the rule VAL_BUFFER and VAL_TEXT_BUILDER have always
   405	 * used — they hold raw bytes/doubles, so there were never child Values to
   406	 * recurse into — and #685 extends it to the two types that could. A dict
   407	 * holding a 5-element list now measures exactly as one holding a 5-element
   408	 * buffer.
   409	 *
   410	 * Consequences, all of them the point:
   411	 *  - cost is O(own size), never O(reachable), so a short-lived binding that
   412	 *    references a large structure is no longer quadratic (#685: 4.93s -> 0.06s
   413	 *    at N=8000, and linear rather than merely faster);
   414	 *  - cycles and shared substructure are STRUCTURALLY unreachable rather than
   415	 *    defended against, which is why #571's visited set and the depth-64 cap
   416	 *    are gone rather than ported;
   417	 *  - the promise shrinks to one the runtime can keep. Deep semantics claimed a
   418	 *    container's entropy reflected everything it reached, but nothing
   419	 *    re-observed a binding when a referenced structure was mutated, so that
   420	 *    claim was already broken on the first in-place write (measured: 6.2M
   421	 *    mutations onto already-summarized values per fleet run). */
